CREATE TABLE "recipes" (
	"id" serial PRIMARY KEY,
	"item_id" integer CONSTRAINT "recipes_item_id_key" UNIQUE,
	"category" text,
	"title" text NOT NULL,
	"mixing_method" text,
	"dough_temp" text,
	"specific_gravity" text,
	"oven_temp" text,
	"baking_time" text,
	"note" text,
	"mixing_type" text,
	"is_warmed" text,
	"is_sacrifice" text,
	"videos" jsonb DEFAULT '[]',
	"summary" text,
	"detail" text,
	"created_at" timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
	"is_summary_published" boolean DEFAULT false,
	"is_detail_published" boolean DEFAULT false,
	"oven_temp2" text,
	"baking_time2" text
);
CREATE UNIQUE INDEX "recipes_item_id_key" ON "recipes" ("item_id");
CREATE UNIQUE INDEX "recipes_pkey" ON "recipes" ("id");
